cmake_minimum_required(VERSION 3.20)
project(wigglinde_rvsvulkan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT HOVITRON_SOURCE_DIR)
  message(FATAL_ERROR "Set HOVITRON_SOURCE_DIR to the pinned upstream checkout")
endif()
set(core "${HOVITRON_SOURCE_DIR}/RVSVulkan/RVSVulkan")
find_path(VULKAN_INCLUDE_DIR vulkan/vulkan.hpp REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
set(names VulkanContext Shader VulkanWrapper VulkanDrawing VulkanRenderPassAbstract VulkanRenderPass VulkanPipelineAbstract VulkanPipelineWarping VulkanPipelineBlending BuffersControllerAbstract BuffersControllerWarping BuffersControllerBlending VulkanHelperFunction RenderingParameters)
# Upstream's headless copy branch is unreachable: getAttachmentSize() is always
# at least one, also without a display. Keep that render-slot count and select
# the output route with the existing, explicit swapchain flag instead.
file(READ "${core}/SourcesVulkan/VulkanDrawing.cpp" drawing)
set(old_route "if (wrapper->getAttachmentSize() || window->useOpenXR())")
set(new_route "if (window->isSwapChainNeeded() || window->useOpenXR())")
string(FIND "${drawing}" "${old_route}" route_pos)
if(route_pos EQUAL -1)
  message(FATAL_ERROR "Pinned RVSVulkan output routing no longer matches")
endif()
string(REPLACE "${old_route}" "${new_route}" drawing "${drawing}")
string(PREPEND drawing "// Wigglinde change, 23 October 2026: select the headless copy by isSwapChainNeeded(), not the internal render-slot count.\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/VulkanDrawing.cpp" "${drawing}")
foreach(name IN LISTS names)
  if(name STREQUAL "VulkanDrawing")
    list(APPEND sources "${CMAKE_CURRENT_BINARY_DIR}/VulkanDrawing.cpp")
  else()
    list(APPEND sources "${core}/SourcesVulkan/${name}.cpp")
  endif()
endforeach()
add_library(rvsvulkan_core STATIC "${core}/WindowAbstract.cpp" ${sources})
target_include_directories(rvsvulkan_core PUBLIC "${core}" "${core}/SourcesVulkan" "${HOVITRON_SOURCE_DIR}/RVSVulkan/external" "${VULKAN_INCLUDE_DIR}" "${JSON_INCLUDE_DIR}")
target_compile_definitions(rvsvulkan_core PUBLIC NDEBUG NOMINMAX _USE_MATH_DEFINES)
if(MSVC)
  target_compile_definitions(rvsvulkan_core PUBLIC WIN32 VK_USE_PLATFORM_WIN32_KHR)
  target_compile_options(rvsvulkan_core PUBLIC /O2 /utf-8 /FI"${CMAKE_CURRENT_SOURCE_DIR}/portability.h")
else()
  target_compile_options(rvsvulkan_core PUBLIC -O3 -include "${CMAKE_CURRENT_SOURCE_DIR}/portability.h")
  target_link_libraries(rvsvulkan_core PUBLIC ${CMAKE_DL_LIBS})
endif()
add_executable(wigglinde_rvsvulkan stream.cpp)
target_link_libraries(wigglinde_rvsvulkan PRIVATE rvsvulkan_core)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(rvsvulkan_stream_check check_stream.cpp)
  add_test(NAME rvsvulkan_binary_frame_stream COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../check_stream.py" --producer $<TARGET_FILE:rvsvulkan_stream_check>)
  add_executable(rvsvulkan_output_check check_output.cpp)
  target_link_libraries(rvsvulkan_output_check PRIVATE rvsvulkan_core)
  add_test(NAME rvsvulkan_output_routing COMMAND rvsvulkan_output_check)
endif()
# Vulkan-Hpp uses upstream's dynamic loader. Only the graphics driver's Vulkan
# runtime is needed at execution; no SDK, headset, camera DLL or CUDA runtime.
