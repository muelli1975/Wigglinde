cmake_minimum_required(VERSION 3.20)
project(wigglinde_rvs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
if(NOT RVS_SOURCE_DIR)
  message(FATAL_ERROR "Set RVS_SOURCE_DIR to the pinned RVS checkout")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc photo imgcodecs)
set(names BlendedView PolynomialDepth Config Parameters JsonParser SynthesizedView View image_loading blending inpainting transform Unprojector Projector PerspectiveUnprojector PerspectiveProjector EquirectangularUnprojector EquirectangularProjector PoseTraces SpaceTransformer)
foreach(name IN LISTS names)
  list(APPEND sources "${RVS_SOURCE_DIR}/src/${name}.cpp")
endforeach()
add_executable(wigglinde_rvs stream.cpp ${sources})
target_include_directories(wigglinde_rvs PRIVATE "${RVS_SOURCE_DIR}/src" ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(wigglinde_rvs PRIVATE WITH_OPENGL=0 NOMINMAX _USE_MATH_DEFINES)
target_link_libraries(wigglinde_rvs PRIVATE ${OpenCV_LIBS})
find_package(Threads REQUIRED)
target_link_libraries(wigglinde_rvs PRIVATE Threads::Threads)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(wigglinde_rvs PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_sources(wigglinde_rvs PRIVATE app.manifest)
  target_compile_options(wigglinde_rvs PRIVATE /O2 /Gy /utf-8)
  target_link_options(wigglinde_rvs PRIVATE /OPT:REF)
else()
  target_compile_options(wigglinde_rvs PRIVATE -O3 -ffunction-sections -fdata-sections)
  if(APPLE)
    target_link_options(wigglinde_rvs PRIVATE -Wl,-dead_strip)
  else()
    target_link_options(wigglinde_rvs PRIVATE -Wl,--gc-sections)
  endif()
endif()

# Separate executable keeps the validated CPU path independent of GL globals.
option(WITH_RVS_OPENGL "Build the experimental OpenGL backend" OFF)
if(WITH_RVS_OPENGL)
  find_package(OpenGL REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(GLFW_SOURCE_DIR)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${GLFW_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/glfw")
  else()
    find_package(glfw3 3.3 REQUIRED)
  endif()
  if(NOT GLM_INCLUDE_DIR)
    find_path(GLM_INCLUDE_DIR glm/glm.hpp REQUIRED)
  endif()
  set(patched "${CMAKE_CURRENT_BINARY_DIR}/opengl-sources")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_opengl.py"
    "${RVS_SOURCE_DIR}/src" "${patched}" COMMAND_ERROR_IS_FATAL ANY)
  set(gl_sources ${sources})
  foreach(name SynthesizedView BlendedView)
    list(REMOVE_ITEM gl_sources "${RVS_SOURCE_DIR}/src/${name}.cpp")
    list(APPEND gl_sources "${patched}/${name}.cpp")
  endforeach()
  foreach(name helpersGL gl_core_4.5 RFBO)
    list(APPEND gl_sources "${RVS_SOURCE_DIR}/src/${name}.cpp")
  endforeach()
  add_executable(wigglinde_rvs_gl stream.cpp ${gl_sources} "${patched}/Shader.cpp")
  target_include_directories(wigglinde_rvs_gl PRIVATE "${RVS_SOURCE_DIR}/src" ${OpenCV_INCLUDE_DIRS} "${GLM_INCLUDE_DIR}")
  target_compile_definitions(wigglinde_rvs_gl PRIVATE WITH_OPENGL=1 WIGGLINDE_OPENGL=1 WITH_RENDERDOC=0 NOMINMAX _USE_MATH_DEFINES)
  target_link_libraries(wigglinde_rvs_gl PRIVATE ${OpenCV_LIBS} glfw OpenGL::GL Threads::Threads)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(wigglinde_rvs_gl PRIVATE OpenMP::OpenMP_CXX)
  endif()
  if(WIN32)
    target_link_libraries(wigglinde_rvs_gl PRIVATE gdi32 user32)
  else()
    find_package(X11 REQUIRED)
    target_link_libraries(wigglinde_rvs_gl PRIVATE ${X11_LIBRARIES} ${CMAKE_DL_LIBS})
  endif()
  if(MSVC)
    target_sources(wigglinde_rvs_gl PRIVATE app.manifest)
    target_compile_options(wigglinde_rvs_gl PRIVATE /O2 /Gy /utf-8)
    target_link_options(wigglinde_rvs_gl PRIVATE /OPT:REF)
  else()
    target_compile_options(wigglinde_rvs_gl PRIVATE -O3)
  endif()
endif()
