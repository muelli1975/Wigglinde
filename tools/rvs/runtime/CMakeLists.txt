cmake_minimum_required(VERSION 3.20)
project(wigglinde_rvs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
if(NOT RVS_SOURCE_DIR)
  message(FATAL_ERROR "Set RVS_SOURCE_DIR to the pinned RVS checkout")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc photo imgcodecs)
set(names BlendedView PolynomialDepth Config Parameters JsonParser SynthesizedView View image_loading blending inpainting transform Unprojector Projector PerspectiveUnprojector PerspectiveProjector EquirectangularUnprojector EquirectangularProjector PoseTraces SpaceTransformer)
foreach(name IN LISTS names)
  list(APPEND sources "${RVS_SOURCE_DIR}/src/${name}.cpp")
endforeach()
add_executable(wigglinde_rvs stream.cpp ${sources})
target_include_directories(wigglinde_rvs PRIVATE "${RVS_SOURCE_DIR}/src" ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(wigglinde_rvs PRIVATE WITH_OPENGL=0 NOMINMAX _USE_MATH_DEFINES)
target_link_libraries(wigglinde_rvs PRIVATE ${OpenCV_LIBS})
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(wigglinde_rvs PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_sources(wigglinde_rvs PRIVATE app.manifest)
  target_compile_options(wigglinde_rvs PRIVATE /O2 /Gy /utf-8)
  target_link_options(wigglinde_rvs PRIVATE /OPT:REF)
else()
  target_compile_options(wigglinde_rvs PRIVATE -O3 -ffunction-sections -fdata-sections)
  target_link_options(wigglinde_rvs PRIVATE -Wl,--gc-sections)
endif()
